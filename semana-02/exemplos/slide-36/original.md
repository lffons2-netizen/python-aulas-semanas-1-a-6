soma = 0
# Começa em 2, vai até 101 (para incluir o 100) e
pula de 2 em 2
for i in range(2, 101, 2):
    soma += i  # O mesmo que: soma = soma + i

print(f"A soma dos números pares de 1 a 100 é:
{soma}")
