class Quadrado(Forma):
    def __init__(self, lado):
        self.lado = lado
    def area(self):
        return self.lado ** 2
    def perimetro(self):
        return 4 * self.lado

c, q = Circulo(5), Quadrado(4)
print(round(c.area(), 2))   # 78.54
print(q.area())             # 16
